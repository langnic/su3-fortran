\section*{QCD auf dem Gitter}

\begin{frame}
    \frametitle{Einführung: QCD auf dem Gitter}
    \begin{itemize}
        \item Bietet Möglichkeit nicht-störungstheoretischer Berechnungen in QCD
        \item Regularisierung durch diskretisierte Raumzeit
        \item Gitterkonstante entspricht der Cut-off Energie im Impulsraum
        \item Niedrige Energieskalen zugänglich, bspw. Hadronenmassen
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Das Pfadintegral}
    \begin{itemize}
    	\item Pfadintegralformalismus:
    	\begin{equation}
			Z = \int \mathcal{D}\Phi e^{iS[\Phi]}
		\end{equation}
    	\item Durch Einführung imaginärer Zeitkoordinate (Wick-Rotation):\\
    	$\Rightarrow$ Minkowski-Metrik $\rightarrow$ euklidische Metrik\\
    	$\Rightarrow$ komplexe Phase $\rightarrow$ Boltzmann-Faktor\\
    	$\Rightarrow$ Verbindung von Quantenfeldtheorie und stat. Mechanik
    	\item Diskretisierung der Raumzeit: 4-dimensionales Gitter
        \item Diskretisiertes Pfadintegral im euklidischen Raum:
        \begin{equation}
			Z = \int \mathcal{D}[\Phi] e^{-S_E[\Phi]}
		\end{equation}
		Integrationsmaß:
		\begin{equation}
			\mathcal{D}[\Phi] = \prod_{n \in \Lambda} d\Phi(n)
		\end{equation}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{QCD: Gluon Wirkung}
    \begin{itemize}
    	\item Hier: Beschränkung auf Eichfelder / keine Fermionen (quenched QCD)
        \item Einführung von Eich-Links zwischen den Gitterpunkten
        \item Gluon Wirkung:
        \begin{equation}
\label{eq:plaq}
S_G[U] = \frac {\beta} 3 \sum_{n \in \Lambda} \sum_{\mu < \nu} \Re\ \text{tr} [\mathbb{1} - U_{\mu \nu}(n)]
\end{equation}
		\item Plakette:
		\begin{equation}
			U_{\mu \nu}(n) = U_{\mu}(n) U_{\nu}(n+\hat{\mu}) U_{\mu}(n+\hat{\nu})^{\dagger} U_{\nu}(n)^{\dagger}
		\end{equation}
		\item Erwartungswerte von Operatoren:
		\begin{equation}
			\langle O \rangle = \frac 1 Z \int \mathcal{D}[U] e^{-S_G[U]} O[U]
		\end{equation}
		\item $4N$-dimensionales Integral $\Rightarrow$ keine exakte Berechnung möglich
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Monte-Carlo Integration}
    \begin{itemize}
        \item Approximation des Integrals durch ''Stichprobenmittelwert'' zufällig generierter Funktionswerte des Integranden
        \item Zufällige Konfigurationen sollten der Wahrscheinlichkeitsdichte
        \begin{equation}
dP(U) = \frac{e^{-S[U]}\mathcal{D}[U]}{\int \mathcal{D}[U] e^{-S[U]}}
\end{equation}
zufolge generiert werden (Importance-Sampling)
		\item Erwartungswerte:
		\begin{equation}
\langle O \rangle = \lim_{N \rightarrow \infty} \frac{1}{N} \sum_{n = 1}^N O[U_n]
\end{equation}
		\item Wie können solche Konfigurationen erzeugt werden?
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Metropolis-Algorithmus}
    \begin{itemize}
        \item Metropolis-Algorithmus: implementiert Markov-Kette zur iterativen Erzeugung der Konfigurationen durch Updates der alten Konfiguration
        \item Kurzformulierung:
        \begin{enumerate}
        	\item Starte bei beliebiger Konfiguration, wähle einen Link $U_{\mu}(n)$ und erzeuge Updatekandidaten $U_{\mu}(n)'$
        	\item Berechne die Summe der \emph{Staples}
        	\begin{align}
				A = \sum_{i=1}^6 P_i = \sum_{\nu \neq \mu} \big( &U_{\nu}(n+\hat{\mu}) U_{\mu}(n+\hat{\nu})^{\dagger} U_{\nu}(n)^{\dagger}\\
		+ &U_{\nu}(n+\hat{\mu}-\hat{\nu})^{\dagger} U_{\mu}(n-\hat{\nu})^{\dagger} U_{\nu}(n-\hat{\nu}) \big)
			\end{align}
			und damit die Wirkungsänderung
			\begin{equation}
				\Delta S = -\frac{\beta}{N} \Re\ \text{tr} [ (U_{\mu}(n)' - U_{\mu}(n)) A]
			\end{equation}
			\item Akzeptiere Link-Update, falls $r \leq \exp(-\Delta S)$, wobei $r \in [0,1)$ Zufallszahl
			\item Wiederhole diese Schritte mit der neuen Konfiguration
        \end{enumerate}
    \end{itemize}
\end{frame}



\section*{Implementierung}


\begin{frame}[fragile]
    \frametitle{Einschub: Fortran Coarrays}
    \begin{itemize}
        \item Bestandteil des Fortran 2008-Standards
        \item Variablen mit \emph{Codimension}
        \item \emph{Images: } parallel ausgeführte Programminstanzen
        \item Beispiel: \lstinline[basicstyle=\normalsize]{real :: x(5)[*]}{}\\
        	$*$ ist ein Platzhalter für die Anzahl der Images, wird zur Laufzeit festgelegt
        \item Unterstützung in GNU-Fortran 5 - für mehrere Images wird Bibliothek für Datenaustausch benötigt
        \item \emph{OpenCoarrays}\footnotemark ist eine solche
        \item Einschränkungen: u.a. keine Coarray-Pointer, keine Coarray-Returnvalues
    \end{itemize}
    \footnotetext{OpenCoarrays: Open-source Transport Layers Supporting Coarray Fortran Compilers}
\end{frame}

\begin{frame}
    \frametitle{Konzept}
    \begin{itemize}
        \item Ziele:
        \begin{itemize}
        	\item Implementierung der Quenched-QCD mit Coarrays
        	\item einfach zu benutzende API - globale ''Feldperspektive''        	
		\end{itemize}
		\item Umsetzung durch:
        \begin{itemize}
        	\item Typen (Klassen) für Felder und Feldvariablen die als Wrapper für (Co-)Arrays fungieren
        	\item überladene Operatoren für arithmetische Operationen
        	\item Select- und Shiftoperatoren für Felder, um Linkrichtung/-parität und Versatz festzulegen
        	\item Parallel ausgeführte Updates/Operationen auf ausgewählten Links
        	\item Günstige Speicherreihenfolge der Matrizen
		\end{itemize} 
    \end{itemize}
\end{frame}

\begin{frame}[fragile]

function select (a, b) result (tmp)
    class(gauge_field), intent(in) :: a
    integer, dimension(2), intent(in) :: b
    type(gauge_field) :: tmp
        
    call a%copy_shift(tmp, 0, 0)
    tmp%par = b(1)
    tmp%dir = b(2)
    tmp%tmp = .true.
end function
\end{lstlisting}
\end{minipage}
    \hfill
    \begin{minipage}{0.48\linewidth}
        \begin{itemize}
        	\item Wählt eine Linkrichtung und Parität aus
        	\item Akzeptiert \emph{gauge\_field} als ersten und integer array als zweiten Operanden
        	\item Parität definiert als $p = (x+y+z+t)\ mod\ 2 \in \{0, 1\}$
        	\item $par = -1$ bedeutet alle sites
        	\item $dir = 0$ bedeutet alle Richtungen
        \end{itemize}
    \end{minipage}
\end{frame}

\subsection*{Metropolis Algorithmus}

\begin{frame}[fragile]
    \frametitle{Metropolismodul: Staple-Funktion}
    \begin{minipage}{0.63\linewidth}
    \begin{lstlisting}
    call staple%init_zero(s)
    do i = 1, 4
        if (i /= d) then
            call staple%increment( &
                (u.sel.[p,i]) &
                * (u.shift.[i,1].sel.[p,d]) &
                * (u.shift.[d,1].sel.[p,-i]) &
                + (u.shift.[i,-1].sel.[p,-i]) &
                * (u.shift.[i,-1].sel.[p,d]) &
                * (u.shift.[i,-1].shift.[d,1].sel.[p,i]) &
                )
        end if
    end do
    \end{lstlisting}
    \end{minipage}
    \hfill
    \begin{minipage}{0.33\linewidth}
        \begin{itemize}
        	\item Anwendung von Select und Shiftoperatoren
        	\item Benutzt \emph{increment}-Funktion, die für Feldvariablen definiert ist
        \end{itemize}
    \end{minipage}
\end{frame}

\begin{frame}[fragile]
    \frametitle{Metropolismodul: Metro-Routine}
    \begin{lstlisting}
do p = 1,2
    do d = 1,4
        stp = staple(u,p,d)	! calc. staple
        stp%conj = .true.
        call random_number(rnd1)	! generate new link
        call random_number(rnd2)
        if (rnd1 >= 0.5) then
            update_tab(ceiling(rnd2*100))%conj = .true.
        else
            update_tab(ceiling(rnd2*100))%conj = .false.
        end if
        new_link = (u.sel.[p,d]) * update_tab(ceiling(rnd2*100))
        delta_s = -beta / 3.0 * .rtrc.((new_link - (u.sel.[p,d])) * stp)
        call rnd_field%init_rnd(s)	! accept with some prob.
        call u%update(p, d, new_link, (rnd_field <= exp(-delta_s)), acc_count)
        sync all
    end do
end do
	\end{lstlisting}
\end{frame}
